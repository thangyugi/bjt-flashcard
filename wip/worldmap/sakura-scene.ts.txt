/** Dựng cảnh 3D của "Làng Sakura bên sông" từ bố cục (sakura.ts) — chuyển từ bản mẫu map-a */
import * as THREE from "three";
import { mergeGeometries } from "three/examples/jsm/utils/BufferGeometryUtils.js";
import { sceneBase } from "../base";
import { Builder, G, MT, clearPrimitives, gridRect, setLayer, type AddOpts } from "../builder";
import { lerp, segDist, smooth, type P2 } from "../core";
import { Drifters, Puffs, Train, passSchedule } from "../life";
import { TINT_A } from "../materials";
import { WalkMap, type Bridge, type Circle, type ORect } from "../nav";
import { INK_A, PAPER_A } from "../post";
import { paintBloom, paintStoneBlocks, paintTiles } from "../textures";
import type { BuildCtx, MapWorld, Zone } from "../types";
import { fallSheet, ribbon, waterMat } from "../water";
import { createSakuraLayout, SAKURA_SEED } from "./sakura";
import { texFrom } from "../builder";

export async function buildSakura(ctx: BuildCtx): Promise<MapWorld> {
  const L = createSakuraLayout();
  const {
    W, ZR, rx, rxd, LK, lakeE, zc, isHigh, ZF, rw, riverD, TC, terr, paddyAt, SH, STREAM, streamY, SPAN0, SPAN1, railY,
    hHigh, hLow, gy, PLOTS, TULIP, RAPE, AZB, inRect, inPlot, ORCH1, ORCH2, STREET, STATION, MACHIYA, TEA, FROG, POOL, PICNIC,
    BRIDGE_Z, lanePts, onPath, onStreet, nearRail, wet, occ, free, TREES, okLand,
  } = L;
  const B0 = sceneBase(ctx, {
    seed: SAKURA_SEED,
    bg: "#e8dcc4",
    tint: TINT_A,
    sun: { pos: [-22, 40, 18], box: [-58, 58, 50, -50], far: 140 },
  });
  const { scene, rand, T, sign } = B0;
  const { R, rng, pick } = rand;
  const U = ctx.U;
  const bloomTex = texFrom(paintBloom(rand, "oval") as HTMLCanvasElement, true, true);
  const tileTex = texFrom(paintTiles(rand, "rib") as HTMLCanvasElement, true, true);
  const blockTex = texFrom(paintStoneBlocks(rand) as HTMLCanvasElement, true, true);
  const kit = B0.makeKit({
    prop: { map: T.grain, vertexColors: true },
    wood: { map: T.wood, vertexColors: true },
    leaf: { map: T.leaf, vertexColors: true },
    bloom: { map: bloomTex, vertexColors: true },
    fur: { map: T.fur, vertexColors: true },
    stone: { map: T.stone, vertexColors: true },
    tile: { map: tileTex, vertexColors: true },
    block: { map: blockTex, vertexColors: true },
    dbl: { map: T.grain, vertexColors: true, side: THREE.DoubleSide },
  });
  const SB = new Builder(kit, rand); // mọi vật tĩnh
  const TB = new Builder(kit, rand, true); // cây (đung đưa)
  const at = (x: number, z: number, rot = 0, dy = 0) => new THREE.Matrix4().makeRotationY(rot).setPosition(x, gy(x, z) + dy, z);
  const circles: Circle[] = [],
    rects: ORect[] = [],
    bridges: Bridge[] = [];
  const zones: Zone[] = [];

  /* ---------- địa hình + vách cao nguyên ---------- */
  const groundMat = kit.paint({ map: T.ground });
  const ground = new THREE.Mesh(gridRect(-66, 66, -70, 56, Math.round(420 * ctx.q.mesh), Math.round(400 * ctx.q.mesh), W, gy), groundMat);
  ground.receiveShadow = true;
  scene.add(ground);
  await ctx.step(0.1);
  const CL = (() => {
    const raw: P2[] = [];
    for (let x = 2.5; x <= 70; x += 0.02) raw.push([x, zc(x)]);
    const out: P2[] = [raw[0]];
    let acc = 0;
    for (let i = 1; i < raw.length; i++) {
      acc += Math.hypot(raw[i][0] - raw[i - 1][0], raw[i][1] - raw[i - 1][1]);
      if (acc >= 0.4) {
        out.push(raw[i]);
        acc = 0;
      }
    }
    return out;
  })();
  {
    const pos: number[] = [],
      uv: number[] = [],
      idx: number[] = [],
      ROWS = 18,
      seed = 51,
      lean = 0.9;
    let arc = 0;
    const fbm2 = (await import("../core")).fbm;
    for (let i = 0; i < CL.length; i++) {
      const [x, z] = CL[i],
        [xa, za] = CL[Math.max(0, i - 1)],
        [xb, zb] = CL[Math.min(CL.length - 1, i + 1)];
      if (i) arc += Math.hypot(x - xa, z - za);
      let tx = xb - xa,
        tz = zb - za;
      const Ln = Math.hypot(tx, tz);
      tx /= Ln;
      tz /= Ln;
      const nx = -tz,
        nz = tx;
      const top = hHigh(x - nx * 0.3, z - nz * 0.3),
        bot = hLow(x + nx * 1.5, z + nz * 1.5) - 0.7;
      for (let r = 0; r <= ROWS; r++) {
        const k = r / ROWS,
          y = lerp(top, bot, k);
        let off = 0;
        if (r > 0) {
          const rough = (fbm2(arc * 0.32, y * 0.5, 3, seed) - 0.5) * 2;
          const lv = y * 0.62 + fbm2(arc * 0.1, 0, 1, seed + 3) * 0.6,
            ledge = Math.pow(lv - Math.floor(lv), 2.5);
          off = lean * k + 0.6 * rough * Math.min(1, k * 5) + 0.32 * ledge * Math.min(1, k * 5);
        }
        pos.push(x + nx * off, y, z + nz * off);
        uv.push(arc / 7, 1 - k);
      }
    }
    for (let i = 0; i < CL.length - 1; i++)
      for (let r = 0; r < ROWS; r++) {
        const a = i * (ROWS + 1) + r,
          b = (i + 1) * (ROWS + 1) + r;
        idx.push(a, b, a + 1, b, b + 1, a + 1);
      }
    const g = new THREE.BufferGeometry();
    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
    g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
    g.setIndex(idx);
    g.computeVertexNormals();
    const m = new THREE.Mesh(g, kit.paint({ map: T.cliff, side: THREE.DoubleSide }));
    m.castShadow = true;
    m.receiveShadow = true;
    scene.add(m);
  }

  /* ---------- nước ---------- */
  const riverMat = waterMat(U, T.water, { speed: 0.32 });
  const WY = -0.12;
  const riverPts = (z0: number, z1: number, y: number) => {
    const p: { x: number; z: number; w: number; y: number }[] = [];
    for (let z = z0; z <= z1 + 1e-6; z += 0.35) p.push({ x: rx(z), z, w: rw(z) + 0.35, y });
    return p;
  };
  scene.add(ribbon(riverPts(-62, ZF + 0.2, 4.85), riverMat));
  scene.add(ribbon(riverPts(ZF + 0.6, LK.z - 2, WY), riverMat));
  {
    const p: { x: number; z: number; w: number; y: number }[] = [];
    const n = (STREAM.length - 1) * 5 + 1;
    for (let i = 0; i < STREAM.length - 1; i++) {
      const [ax, az] = STREAM[i],
        [bx, bz] = STREAM[i + 1];
      for (let t = 0; t < 1 - 1e-6; t += 0.2) p.push({ x: lerp(ax, bx, t), z: lerp(az, bz, t), w: 0.62, y: 0 });
    }
    const last = STREAM[STREAM.length - 1];
    p.push({ x: last[0], z: last[1], w: 0.62, y: 0 });
    p.forEach((q, i) => (q.y = streamY(i / (n - 1)) + 0.02));
    scene.add(ribbon(p, waterMat(U, T.water, { speed: 0.5 }), 1.6));
  }
  {
    const g = new THREE.CircleGeometry(1, 64, 0, Math.PI * 2);
    const p = g.attributes.position,
      uv = g.attributes.uv;
    for (let i = 0; i < p.count; i++) {
      const x = p.getX(i),
        y = p.getY(i),
        r = Math.hypot(x, y);
      const X = LK.x + x * LK.a * 1.12,
        Z = LK.z - y * LK.b * 1.12;
      p.setXYZ(i, X, WY, Z);
      uv.setXY(i, 0.5 + 0.5 * Math.min(1, r), (X * 0.55 + Z * 0.35) / 3.2);
    }
    g.computeVertexNormals();
    scene.add(new THREE.Mesh(g, waterMat(U, T.water, { speed: 0.05 })));
  }
  {
    const zEdge = ZF,
      yTop = 4.86,
      yBot = -0.4,
      reach = 1.6;
    const s = rxd(zEdge),
      cx = rx(zEdge),
      wd = (rw(zEdge - 0.5) + 0.3) * Math.sqrt(1 + s * s),
      H = yTop - yBot;
    scene.add(
      fallSheet((u, t) => {
        const wob = (u - 0.5) * 2 * (0.12 * Math.sin(t * 7 + u * 3) + 0.08 * Math.sin(t * 13 + 1));
        const x = cx + (u - 0.5) * 2 * wd * (1 + 0.22 * t) + wob + Math.sin(u * 9 + t * 3) * 0.05 + s * (reach * Math.pow(t, 0.65));
        const bulge = 0.25 * Math.sin(u * Math.PI) * (1 - t);
        const y = yTop - H * t,
          z = zc(x) + 0.35 + bulge + reach * Math.pow(t, 0.65) - 0.35 * Math.pow(1 - t, 6);
        return [x, y, z];
      }, waterMat(U, T.water, { speed: 1.2, fall: 1, vScale: H / 0.6 }))
    );
  }
  const puffs = new Puffs(scene, T.soft, rand);
  puffs.add(rx(ZF + 1.9), -0.25, ZF + 2.0, rw(ZF) + 0.8, 0.6, 30, 1.5, 0.9, 0.85);
  for (let i = 0; i < 6; i++) {
    const s = new THREE.Sprite(new THREE.SpriteMaterial({ map: T.soft, color: 0xf4fbff, transparent: true, depthWrite: false, opacity: 0.25 }));
    s.position.set(rx(ZF + 2) + R(-2.5, 2.5), R(0.5, 2.0), ZF + R(2.0, 3.5));
    s.scale.setScalar(R(3, 4.5));
    setLayer(s, 1);
    scene.add(s);
  }
  // cánh hoa trôi trên sông + hồ
  {
    const pg = new THREE.CircleGeometry(0.075, 5);
    pg.rotateX(-Math.PI / 2);
    pg.scale(1, 1, 0.65);
    const N = Math.round(2600 * ctx.q.scatter),
      im = new THREE.InstancedMesh(pg, new THREE.MeshBasicMaterial({ color: 0xffffff }), N),
      m4 = new THREE.Matrix4();
    const cols = ["#ffd6e0", "#ffc2d2", "#ffe9ef", "#ffffff", "#f7aec4"].map((c) => new THREE.Color(c));
    let c = 0;
    for (let i = 0; i < 40000 && c < N; i++) {
      const z = R(ZF + 3, LK.z + 7),
        x = rx(Math.min(z, LK.z - 2)) + R(-4, 4);
      let ok = false;
      const y = WY + 0.02;
      if (z < LK.z - 2 && riverD(x, z) < rw(z) - 0.15) ok = rng() < 0.75 + 0.25 * (riverD(x, z) / rw(z));
      const xl = LK.x + R(-LK.a, LK.a),
        zl = LK.z + R(-LK.b, LK.b);
      if (!ok && lakeE(xl, zl) < 0.98 && rng() < 0.35 + 0.65 * smooth(0.5, 0.95, lakeE(xl, zl))) {
        ok = true;
        m4.makeRotationY(R(0, 6)).setPosition(xl, y, zl);
      } else if (ok) m4.makeRotationY(R(0, 6)).setPosition(x, y, z);
      if (!ok) continue;
      if (rng() < 0.4) {
        const e = m4.elements;
        e[12] += R(-0.15, 0.15);
        e[14] += R(-0.15, 0.15);
      }
      im.setMatrixAt(c, m4);
      im.setColorAt(c, pick(cols));
      c++;
    }
    im.count = c;
    setLayer(im, 1);
    scene.add(im);
  }
  await ctx.step(0.2);

  /* ---------- đường sắt: cầu cạn, ray, ga ---------- */
  const RAIL_G = 0.42;
  {
    const b = SB;
    const nSp = 7,
      spw = (SPAN1 - SPAN0) / nSp,
      depth = 1.9,
      pw = 0.55;
    for (let i = 0; i < nSp; i++) {
      const x0 = SPAN0 + i * spw,
        x1 = x0 + spw,
        yt = railY((x0 + x1) / 2) - 0.2,
        yb = -1.4;
      const r = (spw - 2 * pw) / 2,
        ys = yt - 0.42 - r;
      const sh = new THREE.Shape();
      sh.moveTo(x0, yb);
      sh.lineTo(x0 + pw, yb);
      sh.lineTo(x0 + pw, ys);
      sh.absarc((x0 + x1) / 2, ys, r, Math.PI, 0, true);
      sh.lineTo(x1 - pw, yb);
      sh.lineTo(x1, yb);
      sh.lineTo(x1, yt);
      sh.lineTo(x0, yt);
      sh.closePath();
      const g = new THREE.ExtrudeGeometry(sh, { depth, bevelEnabled: false, curveSegments: 18 });
      const uv = g.attributes.uv,
        p = g.attributes.position;
      for (let k = 0; k < uv.count; k++) uv.setXY(k, (p.getX(k) + p.getZ(k)) / 2.4, p.getY(k) / 2.4);
      b.add(g, "#f1e9db", { p: [0, 0, ZR - depth / 2], mat: "block" });
      g.dispose();
      b.add(G.box(pw * 2 + 0.3, 0.9, depth + 0.4), "#c9bfae", { p: [x0, gy(x0, ZR) - 0.1, ZR], mat: "block", uv: [0.6, 0.4] });
    }
    b.add(G.box(pw * 2 + 0.3, 0.9, depth + 0.4), "#c9bfae", { p: [SPAN1, gy(SPAN1, ZR) - 0.1, ZR], mat: "block", uv: [0.6, 0.4] });
    for (let x = SPAN0 - 0.2; x < SPAN1 + 0.2; x += 1.0) {
      const y = railY(x + 0.5) - 0.2;
      b.add(G.box(1.02, 0.18, depth + 0.35), "#e2d6c1", { p: [x + 0.5, y + 0.02, ZR], mat: "block", uv: [0.4, 0.1] });
      for (const sz of [-1, 1]) b.add(G.box(0.12, 0.62, 0.12), "#c23b2e", { p: [x + 0.5, y + 0.4, ZR + sz * (depth / 2 + 0.08)] });
    }
    for (const sz of [-1, 1])
      for (let x = SPAN0 - 0.2; x < SPAN1 + 0.2; x += 2.0) {
        const xa = x,
          xb = x + 2.0,
          ya = railY(xa) + 0.42,
          yb2 = railY(xb) + 0.42;
        b.add(G.box(Math.hypot(2.0, yb2 - ya) + 0.06, 0.1, 0.1), "#d4473a", { p: [(xa + xb) / 2, (ya + yb2) / 2, ZR + sz * (depth / 2 + 0.08)], r: [0, 0, Math.atan2(yb2 - ya, 2.0)] });
      }
    for (let x = -66; x <= 66; x += 0.55) b.add(G.box(0.24, 0.1, 1.35), pick(["#6b4a33", "#5d3f2b", "#77543a"]), { p: [x, railY(x) - 0.03, ZR], mat: "wood" });
    for (const sz of [-1, 1])
      for (let x = -66; x < 66; x += 1.0) {
        const ya = railY(x) + 0.06,
          yb2 = railY(x + 1.0) + 0.06;
        b.add(G.box(Math.hypot(1.0, yb2 - ya) + 0.02, 0.09, 0.08), "#8d8f96", { p: [x + 0.5, (ya + yb2) / 2, ZR + sz * RAIL_G], r: [0, 0, Math.atan2(yb2 - ya, 1.0)] });
      }
  }
  rects.push({ x: (SPAN0 + SPAN1) / 2, z: ZR, w: SPAN1 - SPAN0 + 2, d: 2.6 });

  // tàu kem–đỏ, 3 toa, mặt cười ở toa đầu (mũi về −x), mỗi toa một nhóm để chạy
  const L0 = 3.1,
    GAP = 0.28;
  const cars: THREE.Group[] = [];
  for (let k = 0; k < 3; k++) {
    const b = new Builder(kit, rand);
    const M = new THREE.Matrix4();
    const front = k === 0,
      tail = k === 2;
    const cream = "#fbf1dc",
      red = "#d23a32",
      red2 = "#b52e2a",
      glass = "#3d5f7a";
    b.add(G.box(L0 - 0.2, 0.22, 1.1), "#3b3330", { p: [0, 0.2, 0], parent: M });
    for (const wx of [-1.0, 1.0]) for (const sz of [-1, 1]) b.add(G.cyl(0.22, 0.22, 0.12, 12), "#2c2624", { p: [wx, 0.2, sz * 0.5], r: [Math.PI / 2, 0, 0], cap: "#5b514c", parent: M });
    b.add(G.box(L0, 0.5, 1.3), red, { p: [0, 0.58, 0], wob: 0.012, parent: M });
    b.add(G.box(L0, 0.62, 1.3), cream, { p: [0, 1.13, 0], wob: 0.012, parent: M });
    b.add(G.box(L0, 0.08, 1.34), red2, { p: [0, 0.85, 0], parent: M });
    const roof = new THREE.CylinderGeometry(0.7, 0.7, L0, 16, 1, false, -Math.PI / 2, Math.PI);
    b.add(roof, red, { p: [0, 1.44, 0], r: [0, 0, Math.PI / 2], s: [1, 1, 0.42], parent: M });
    roof.dispose();
    b.add(G.box(0.5, 0.12, 0.5), "#e9dcc2", { p: [0.3, 1.78, 0], parent: M });
    for (let i = 0; i < 4; i++)
      for (const sz of [-1, 1]) {
        const wx = -1.12 + i * 0.75;
        b.add(G.box(0.5, 0.36, 0.04), glass, { p: [wx, 1.16, sz * 0.66], parent: M });
        b.add(G.box(0.18, 0.08, 0.05), "#ffffff", { p: [wx - 0.1, 1.27, sz * 0.67], parent: M });
      }
    if (!tail) b.add(G.box(GAP + 0.1, 0.18, 0.4), "#3b3330", { p: [L0 / 2 + GAP / 2, 0.42, 0], parent: M });
    if (front) {
      const N = new THREE.Matrix4().makeTranslation(-L0 / 2, 0, 0);
      const n1 = new THREE.SphereGeometry(0.66, 20, 14, 0, Math.PI),
        n2 = new THREE.SphereGeometry(0.66, 20, 10, 0, Math.PI * 2, Math.PI / 2, Math.PI / 2);
      b.add(n1, cream, { p: [0, 1.05, 0], r: [0, -Math.PI / 2, 0], s: [0.7, 0.95, 1], parent: N });
      b.add(n2, red, { p: [0, 0.62, 0], s: [0.7, 0.6, 1], parent: N });
      n1.dispose();
      n2.dispose();
      for (const sz of [-1, 1]) {
        b.add(G.cyl(0.17, 0.17, 0.08, 16), "#fffbe8", { p: [-0.45, 1.1, sz * 0.26], r: [0, 0, Math.PI / 2], parent: N });
        b.add(G.cyl(0.09, 0.09, 0.05, 12), "#2a2220", { p: [-0.5, 1.1, sz * 0.24], r: [0, 0, Math.PI / 2], parent: N });
        b.add(G.sph(0.03, 6, 5), "#ffffff", { p: [-0.53, 1.14, sz * 0.21], parent: N });
        b.add(G.sph(0.1, 10, 8), "#f49aac", { p: [-0.38, 0.9, sz * 0.44], s: [0.4, 0.6, 1], parent: N });
        b.add(G.cone(0.2, 0.36, 10), red, { p: [0.2, 1.88, sz * 0.38], r: [sz * 0.25, 0, 0], parent: new THREE.Matrix4().makeTranslation(-L0 / 2 + 0.4, 0, 0) });
      }
      b.add(G.tor(0.11, 0.025, 6, 12, Math.PI), "#2a2220", { p: [-0.47, 0.92, 0], r: [0, Math.PI / 2, Math.PI], parent: N });
      b.add(G.box(0.12, 0.05, 0.9), "#e9c34a", { p: [-0.36, 0.42, 0], parent: N });
    }
    if (tail) b.add(G.box(0.08, 0.22, 0.22), "#f5d76e", { p: [L0 / 2 + 0.03, 0.75, 0.38], parent: M });
    const g = b.build(true, false);
    const bl = new THREE.Mesh(new THREE.PlaneGeometry(L0 + 0.6, 1.9), new THREE.MeshBasicMaterial({ map: T.blob, transparent: true, depthWrite: false, opacity: 0.8 }));
    bl.rotation.x = -Math.PI / 2;
    bl.position.y = -0.1;
    setLayer(bl, 1);
    g.add(bl);
    if (front)
      for (let i = 0; i < 4; i++) {
        const s = new THREE.Sprite(new THREE.SpriteMaterial({ map: T.soft, color: 0xffffff, transparent: true, depthWrite: false, opacity: 0.7 - i * 0.12 }));
        s.position.set(0.45 + i * 0.9, 2.06 + i * 0.35, -i * 0.2);
        s.scale.setScalar(0.7 + i * 0.35);
        s.userData.puff = i;
        setLayer(s, 1);
        g.add(s);
      }
    g.matrixAutoUpdate = true;
    for (const c of g.children) {
      c.matrixAutoUpdate = true;
    }
    scene.add(g);
    cars.push(g);
  }
  const SPACING = L0 + GAP;
  const train = new Train(
    cars,
    SPACING,
    (s) => {
      const x = -s;
      const dy = railY(x + 0.5) - railY(x - 0.5);
      return { x, y: railY(x) + 0.14, z: ZR, yaw: 0, pitch: Math.atan2(dy, 1) };
    },
    (s) => Math.abs(s) > 64
  );
  const STOP = -(STATION.x - SPACING);

  /* ---------- nhà cửa ---------- */
  function gableRoof(b: Builder, M: THREE.Matrix4, w: number, d: number, yEave: number, rise: number, color: string, oh = 0.5, ridgeCol = "#3d4656") {
    const half = d / 2 + oh,
      slope = Math.hypot(half, rise),
      ang = Math.atan2(rise, half),
      Wr = w + 2 * oh;
    for (const sz of [-1, 1]) b.add(G.box(Wr, 0.14, slope), color, { p: [0, yEave + rise / 2, (sz * half) / 2], r: [sz * ang, 0, 0], mat: "tile", uv: [Wr / 2, slope / 2], wob: 0.012, parent: M });
    b.add(G.box(Wr + 0.1, 0.2, 0.26), ridgeCol, { p: [0, yEave + rise + 0.05, 0], parent: M });
    const tri = new THREE.Shape();
    tri.moveTo(-d / 2, 0);
    tri.lineTo(d / 2, 0);
    tri.lineTo(0, (rise * (d / 2)) / half);
    tri.closePath();
    const tg = new THREE.ShapeGeometry(tri);
    for (const sx of [-1, 1]) b.add(tg, "#f1e6cf", { p: [((sx * w) / 2) * 0.999, yEave, 0], r: [0, (sx * Math.PI) / 2, 0], parent: M, mat: "dbl" });
    tg.dispose();
    return yEave + rise;
  }

      if (s === null) cars.forEach((c) => (c.visible = false));
      else train.place(s, 1);

        const i = p.userData.puff as number,
          l = (t * 0.8 + i * 0.25) % 1;
        p.position.y = 2.06 + l * 1.4;

      });

    },
